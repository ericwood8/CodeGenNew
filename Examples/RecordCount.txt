CREATE FUNCTION Account_RecordCount() 
RETURNS INT
AS
BEGIN
    -- Purpose: Fast way used by Microsoft to determine record count
    -- rather than doing SELECT COUNT(*) FROM <Table>

    DECLARE @ReturnVal INT;

    SELECT @ReturnVal = SUM(spart.rows)
    FROM sys.partitions spart
    WHERE spart.object_id = object_id('Account')
      AND spart.index_id < 2;

    RETURN @ReturnVal;
END
GO
