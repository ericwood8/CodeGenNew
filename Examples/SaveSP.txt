--  ** Always pass entire record on updates or inserts. **

CREATE PROCEDURE Account_Save 
(   @ID INT OUTPUT,
   @AccountNumber VARCHAR(10),
   @ShortDescr VARCHAR(50),
   @BalanceTypeEnumID INT,
   @FinancialCodeEnumID INT,
   @JIBCashFlowCategoryEnumID INT,
   @RDCashFlowCategoryEnumID INT,
   @IsBillable BIT = 0,
   @ContraAccount_AccountID INT,
   @IsAFEAccount BIT = 0,
   @AFEAccountTypeEnumID INT,
   @AFEAccountCasingPointTypeEnumID INT,
   @AFEExpenseTypeEnumID INT,
   @WellcostAllocationEnumID INT,
   @PropertyTypeEnumID INT,
   @DepletionAccountTypeEnumID INT,
   @AccountType1099EnumID INT,
   @DepreciationAssetTypeEnumID INT,
   @LeaseOperatingStatementEnumID INT,
   @NoteText TEXT 
)
AS
BEGIN
    SET NOCOUNT OFF;
    DECLARE @Count INT;

    IF (@ID IS NULL)
      SET @Count =  0;
    ELSE
    BEGIN
        SELECT @Count = COUNT([ID])
        FROM Account
        WHERE [ID] = @ID;  
    END

    -- Fix strings
    SET @AccountNumber = dbo.uf_FixString(@AccountNumber);
    SET @ShortDescr = dbo.uf_FixString(@ShortDescr);

    BEGIN TRANSACTION;
    BEGIN TRY --  Validate data inside try

    IF (@Count > 0) -- Update if record found
    BEGIN
        UPDATE Account SET
           [AccountNumber] = @AccountNumber,
           [ShortDescr] = @ShortDescr,
           [BalanceTypeEnumID] = @BalanceTypeEnumID,
           [FinancialCodeEnumID] = @FinancialCodeEnumID,
           [JIBCashFlowCategoryEnumID] = @JIBCashFlowCategoryEnumID,
           [RDCashFlowCategoryEnumID] = @RDCashFlowCategoryEnumID,
           [IsBillable] = @IsBillable,
           [ContraAccount_AccountID] = @ContraAccount_AccountID,
           [IsAFEAccount] = @IsAFEAccount,
           [AFEAccountTypeEnumID] = @AFEAccountTypeEnumID,
           [AFEAccountCasingPointTypeEnumID] = @AFEAccountCasingPointTypeEnumID,
           [AFEExpenseTypeEnumID] = @AFEExpenseTypeEnumID,
           [WellcostAllocationEnumID] = @WellcostAllocationEnumID,
           [PropertyTypeEnumID] = @PropertyTypeEnumID,
           [DepletionAccountTypeEnumID] = @DepletionAccountTypeEnumID,
           [AccountType1099EnumID] = @AccountType1099EnumID,
           [DepreciationAssetTypeEnumID] = @DepreciationAssetTypeEnumID,
           [LeaseOperatingStatementEnumID] = @LeaseOperatingStatementEnumID,
           [NoteText] = @NoteText 
        WHERE [ID] = @ID;
    END
    ELSE
    BEGIN
        INSERT INTO Account
         (         [AccountNumber],
         [ShortDescr],
         [BalanceTypeEnumID],
         [FinancialCodeEnumID],
         [JIBCashFlowCategoryEnumID],
         [RDCashFlowCategoryEnumID],
         [IsBillable],
         [ContraAccount_AccountID],
         [IsAFEAccount],
         [AFEAccountTypeEnumID],
         [AFEAccountCasingPointTypeEnumID],
         [AFEExpenseTypeEnumID],
         [WellcostAllocationEnumID],
         [PropertyTypeEnumID],
         [DepletionAccountTypeEnumID],
         [AccountType1099EnumID],
         [DepreciationAssetTypeEnumID],
         [LeaseOperatingStatementEnumID],
         [NoteText]) 
        VALUES
       (         @AccountNumber,
         @ShortDescr,
         @BalanceTypeEnumID,
         @FinancialCodeEnumID,
         @JIBCashFlowCategoryEnumID,
         @RDCashFlowCategoryEnumID,
         @IsBillable,
         @ContraAccount_AccountID,
         @IsAFEAccount,
         @AFEAccountTypeEnumID,
         @AFEAccountCasingPointTypeEnumID,
         @AFEExpenseTypeEnumID,
         @WellcostAllocationEnumID,
         @PropertyTypeEnumID,
         @DepletionAccountTypeEnumID,
         @AccountType1099EnumID,
         @DepreciationAssetTypeEnumID,
         @LeaseOperatingStatementEnumID,
         @NoteText);

        SET @ID = @@IDENTITY;  -- learn the id of the newly inserted record
    END

      COMMIT TRANSACTION;
      RETURN 0; -- Return 0 as answer to SP if all was good.
    END TRY
    BEGIN CATCH
      -- NOTE - SQL Server returns the error number as answer to SP
      ROLLBACK TRANSACTION;
      EXECUTE dbo.spLogError; -- Logs error to table
      THROW;
    END CATCH;
END
GO


CREATE FUNCTION [dbo].[uf_FixString]
(@inVarchar VARCHAR(MAX))
RETURNS VARCHAR(MAX)
AS
BEGIN
  -- PURPOSE: 1) Puts extra single quote if any in string value so SQL Server does not take 
  --   that as string marker instead. 2) Trims leading and trailing spaces so weird mismatches do not occur.
  --   3) Wraps special characters.
  -- Do not need NULL handling on REPLACE or LTRIM or RTRIM commands
  -- WARNING: Optional Parameter Default Values are not triggered on explicit parameter setting.
  --    Say you have SP:  spTest (@Notes VARCHAR(4096) = '')
  --    Say you call it call it:  DECLARE @Notes varchar(4096);
  --                              EXEC spTest @Notes=NULL;
  --    Results are unexpected @Notes variable is NULL and it is not set to ''.
  --  For this reason, we consistently set NULL strings to '' because a string column might be NOT NULL by accident.

  DECLARE @s VARCHAR(MAX);

  IF (@inVarchar IS NULL) 
    SET @s = '';
  ELSE IF (@inVarchar = '')
    SET @s = @inVarchar;
  ELSE
  BEGIN
	SET @s = REPLACE(LTRIM(RTRIM(@inVarchar)), CHAR(39), CHAR(39) + CHAR(39));
	SET @s = REPLACE(REPLACE(@s, '[', '[[]'), '%', '[%]') ;
  END;    
  
  RETURN @s;
END
GO
