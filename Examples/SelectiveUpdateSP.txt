USE ProvidenceOgas
GO
SET ANSI_NULLS ON;
GO
SET QUOTED_IDENTIFIER ON;
GO


CREATE PROCEDURE Account_Update 
--  Purpose: Allows SPs to call with just the data that changed in the parameters. 
--  Set all parameters to NULL to detect what changed so we can do selective updates.  Only in trouble if user can set to NULL.
--  Example Call: EXECUTE dbo.Account_Update @OwnerID=@ID;

(@ID INT = NULL OUTPUT,
   @AccountNumber VARCHAR(10) = NULL,
   @ShortDescr VARCHAR(50) = NULL,
   @BalanceTypeEnumID INT = NULL,
   @FinancialCodeEnumID INT = NULL,
   @JIBCashFlowCategoryEnumID INT = NULL,
   @RDCashFlowCategoryEnumID INT = NULL,
   @IsBillable BIT = NULL,
   @ContraAccount_AccountID INT = NULL,
   @IsAFEAccount BIT = NULL,
   @AFEAccountTypeEnumID INT = NULL,
   @AFEAccountCasingPointTypeEnumID INT = NULL,
   @AFEExpenseTypeEnumID INT = NULL,
   @WellcostAllocationEnumID INT = NULL,
   @PropertyTypeEnumID INT = NULL,
   @DepletionAccountTypeEnumID INT = NULL,
   @AccountType1099EnumID INT = NULL,
   @DepreciationAssetTypeEnumID INT = NULL,
   @LeaseOperatingStatementEnumID INT = NULL,
   @NoteText TEXT = NULL)
AS
BEGIN
    EXECUTE dbo.spLogPath @@PROCID; -- Logs our path through the SPs.

    DECLARE @strUpdate varchar(8000),
            @strSet varchar(8000);

    SET @strSet = '';

    -- Fix strings
    SET @AccountNumber = dbo.uf_FixString(@AccountNumber);
    SET @ShortDescr = dbo.uf_FixString(@ShortDescr);

    BEGIN TRANSACTION;
    BEGIN TRY
    IF @ID IS NOT NULL
    BEGIN
        SET @strUpdate = 'UPDATE Account SET '; 
        IF @AccountNumber IS NOT NULL
        SET @strSet = @strSet + '[AccountNumber] = ' + @AccountNumber + ',';

        IF @ShortDescr IS NOT NULL
        SET @strSet = @strSet + '[ShortDescr] = ' + @ShortDescr + ',';

        IF @BalanceTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[BalanceTypeEnumID] = ' + CONVERT(VARCHAR(50),@BalanceTypeEnumID) + ',';

        IF @FinancialCodeEnumID IS NOT NULL
        SET @strSet = @strSet + '[FinancialCodeEnumID] = ' + CONVERT(VARCHAR(50),@FinancialCodeEnumID) + ',';

        IF @JIBCashFlowCategoryEnumID IS NOT NULL
        SET @strSet = @strSet + '[JIBCashFlowCategoryEnumID] = ' + CONVERT(VARCHAR(50),@JIBCashFlowCategoryEnumID) + ',';

        IF @RDCashFlowCategoryEnumID IS NOT NULL
        SET @strSet = @strSet + '[RDCashFlowCategoryEnumID] = ' + CONVERT(VARCHAR(50),@RDCashFlowCategoryEnumID) + ',';

        IF @IsBillable IS NOT NULL
        SET @strSet = @strSet + '[IsBillable] = ' + CONVERT(VARCHAR(50),@IsBillable) + ',';

        IF @ContraAccount_AccountID IS NOT NULL
        SET @strSet = @strSet + '[ContraAccount_AccountID] = ' + CONVERT(VARCHAR(50),@ContraAccount_AccountID) + ',';

        IF @IsAFEAccount IS NOT NULL
        SET @strSet = @strSet + '[IsAFEAccount] = ' + CONVERT(VARCHAR(50),@IsAFEAccount) + ',';

        IF @AFEAccountTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[AFEAccountTypeEnumID] = ' + CONVERT(VARCHAR(50),@AFEAccountTypeEnumID) + ',';

        IF @AFEAccountCasingPointTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[AFEAccountCasingPointTypeEnumID] = ' + CONVERT(VARCHAR(50),@AFEAccountCasingPointTypeEnumID) + ',';

        IF @AFEExpenseTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[AFEExpenseTypeEnumID] = ' + CONVERT(VARCHAR(50),@AFEExpenseTypeEnumID) + ',';

        IF @WellcostAllocationEnumID IS NOT NULL
        SET @strSet = @strSet + '[WellcostAllocationEnumID] = ' + CONVERT(VARCHAR(50),@WellcostAllocationEnumID) + ',';

        IF @PropertyTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[PropertyTypeEnumID] = ' + CONVERT(VARCHAR(50),@PropertyTypeEnumID) + ',';

        IF @DepletionAccountTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[DepletionAccountTypeEnumID] = ' + CONVERT(VARCHAR(50),@DepletionAccountTypeEnumID) + ',';

        IF @AccountType1099EnumID IS NOT NULL
        SET @strSet = @strSet + '[AccountType1099EnumID] = ' + CONVERT(VARCHAR(50),@AccountType1099EnumID) + ',';

        IF @DepreciationAssetTypeEnumID IS NOT NULL
        SET @strSet = @strSet + '[DepreciationAssetTypeEnumID] = ' + CONVERT(VARCHAR(50),@DepreciationAssetTypeEnumID) + ',';

        IF @LeaseOperatingStatementEnumID IS NOT NULL
        SET @strSet = @strSet + '[LeaseOperatingStatementEnumID] = ' + CONVERT(VARCHAR(50),@LeaseOperatingStatementEnumID) + ',';

        IF @NoteText IS NOT NULL
        SET @strSet = @strSet + '[NoteText] = ' + @NoteText + ',';

        IF LEN(@strSet) > 0
        BEGIN
            -- Get rid of extra comma if is one
            IF RIGHT(@strSet, 1) = ','
              SET @strSet = LEFT(@strSet, LEN(@strSet) -1);

            SET @strUpdate = @strUpdate + @strSet + ' WHERE ([ID] = ' + @ID + ')'; 
            EXECUTE @strUpdate;
        END
    END

      COMMIT TRANSACTION;
      RETURN 0; -- Return 0 as answer to SP if all was good.
    END TRY
    BEGIN CATCH
      -- NOTE - SQL Server returns the error number as answer to SP
      ROLLBACK TRANSACTION;
      EXECUTE dbo.spLogError; -- Logs error to table
      THROW;
    END CATCH;
END
