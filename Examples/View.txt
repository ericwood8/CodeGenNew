CREATE VIEW [dbo].[AccountView] 
AS
SELECT 
        AccountType1099.ShortDescr AS AccountType1099,
        AFEAccountCasingPointType.ShortDescr AS AFEAccountCasingPointType,
        AFEAccountType.ShortDescr AS AFEAccountType,
        AFEExpenseType.ShortDescr AS AFEExpenseType,
        BalanceType.ShortDescr AS BalanceType,
        DepletionAccountType.ShortDescr AS DepletionAccountType,
        DepreciationAssetType.ShortDescr AS DepreciationAssetType,
        FinancialCode.ShortDescr AS FinancialCode,
        JIBCashFlowCategory.ShortDescr AS JIBCashFlowCategory,
        LeaseOperatingStatement.ShortDescr AS LeaseOperatingStatement,
        PropertyType.ShortDescr AS PropertyType,
        RDCashFlowCategory.ShortDescr AS RDCashFlowCategory,
        WellcostAllocation.ShortDescr AS WellcostAllocation,
        Account.ID AS ID,
        Account.AccountNumber AS AccountNumber,
        Account.ShortDescr AS ShortDescr,
        Account.BalanceTypeEnumID AS BalanceTypeEnumID,
        Account.FinancialCodeEnumID AS FinancialCodeEnumID,
        Account.JIBCashFlowCategoryEnumID AS JIBCashFlowCategoryEnumID,
        Account.RDCashFlowCategoryEnumID AS RDCashFlowCategoryEnumID,
        Account.IsBillable AS IsBillable,
        Account.ContraAccount_AccountID AS ContraAccount_AccountID,
        Account.IsAFEAccount AS IsAFEAccount,
        Account.AFEAccountTypeEnumID AS AFEAccountTypeEnumID,
        Account.AFEAccountCasingPointTypeEnumID AS AFEAccountCasingPointTypeEnumID,
        Account.AFEExpenseTypeEnumID AS AFEExpenseTypeEnumID,
        Account.WellcostAllocationEnumID AS WellcostAllocationEnumID,
        Account.PropertyTypeEnumID AS PropertyTypeEnumID,
        Account.DepletionAccountTypeEnumID AS DepletionAccountTypeEnumID,
        Account.AccountType1099EnumID AS AccountType1099EnumID,
        Account.DepreciationAssetTypeEnumID AS DepreciationAssetTypeEnumID,
        Account.LeaseOperatingStatementEnumID AS LeaseOperatingStatementEnumID,
        Account.NoteText AS NoteText 
FROM Account
LEFT OUTER JOIN SYEnum AS AccountType1099 ON AccountType1099.EnumID = Account.AccountType1099EnumID
LEFT OUTER JOIN SYEnum AS AFEAccountCasingPointType ON AFEAccountCasingPointType.EnumID = Account.AFEAccountCasingPointTypeEnumID
LEFT OUTER JOIN SYEnum AS AFEAccountType ON AFEAccountType.EnumID = Account.AFEAccountTypeEnumID
LEFT OUTER JOIN SYEnum AS AFEExpenseType ON AFEExpenseType.EnumID = Account.AFEExpenseTypeEnumID
LEFT OUTER JOIN SYEnum AS BalanceType ON BalanceType.EnumID = Account.BalanceTypeEnumID
LEFT OUTER JOIN Account AS Account ON Account.ID = Account.ContraAccount_AccountID
LEFT OUTER JOIN SYEnum AS DepletionAccountType ON DepletionAccountType.EnumID = Account.DepletionAccountTypeEnumID
LEFT OUTER JOIN SYEnum AS DepreciationAssetType ON DepreciationAssetType.EnumID = Account.DepreciationAssetTypeEnumID
LEFT OUTER JOIN SYEnum AS FinancialCode ON FinancialCode.EnumID = Account.FinancialCodeEnumID
LEFT OUTER JOIN SYEnum AS JIBCashFlowCategory ON JIBCashFlowCategory.EnumID = Account.JIBCashFlowCategoryEnumID
LEFT OUTER JOIN SYEnum AS LeaseOperatingStatement ON LeaseOperatingStatement.EnumID = Account.LeaseOperatingStatementEnumID
LEFT OUTER JOIN SYEnum AS PropertyType ON PropertyType.EnumID = Account.PropertyTypeEnumID
LEFT OUTER JOIN SYEnum AS RDCashFlowCategory ON RDCashFlowCategory.EnumID = Account.RDCashFlowCategoryEnumID
LEFT OUTER JOIN SYEnum AS WellcostAllocation ON WellcostAllocation.EnumID = Account.WellcostAllocationEnumID
GO
